-- 1) Colonnes XP sur les profils
alter table public.profiles add column if not exists xp integer not null default 0;
alter table public.profiles add column if not exists referred_by uuid;

-- 2) Historique des actions (logs XP)
create table if not exists public.xp_logs (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid not null,
  action     text not null,      -- tournament_reg | team_create | avatar_update | checkin | referral
  xp         integer not null,
  ref        text,               -- id tournoi / équipe / filleul / mois
  created_at timestamptz not null default now()
);
create index if not exists xp_logs_user_idx on public.xp_logs(user_id, created_at desc);

-- Anti-doublon (1 seule récompense par action/ref)
create unique index if not exists xp_logs_uniq_tournament on public.xp_logs(user_id, ref) where action = 'tournament_reg';
create unique index if not exists xp_logs_uniq_checkin    on public.xp_logs(user_id, ref) where action = 'checkin';
create unique index if not exists xp_logs_uniq_team       on public.xp_logs(user_id)      where action = 'team_create';
create unique index if not exists xp_logs_uniq_avatar     on public.xp_logs(user_id, ref) where action = 'avatar_update';
create unique index if not exists xp_logs_uniq_referral   on public.xp_logs(user_id, ref) where action = 'referral';

-- 3) Demandes de cadeaux (claim)
create table if not exists public.reward_claims (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid not null,
  tier       text not null,      -- xp_1000 | xp_10000 | xp_20000
  xp         integer not null,
  status     text not null default 'pending',  -- pending | approved | rejected
  created_at timestamptz not null default now(),
  decided_at timestamptz,
  unique(user_id, tier)
);

-- 4) RLS
alter table public.xp_logs       enable row level security;
alter table public.reward_claims enable row level security;

drop policy if exists "xp_logs_read_own_or_admin" on public.xp_logs;
create policy "xp_logs_read_own_or_admin" on public.xp_logs
  for select using (user_id = auth.uid() or public.is_admin());

drop policy if exists "reward_claims_read_own_or_admin" on public.reward_claims;
create policy "reward_claims_read_own_or_admin" on public.reward_claims
  for select using (user_id = auth.uid() or public.is_admin());

drop policy if exists "reward_claims_admin_update" on public.reward_claims;
create policy "reward_claims_admin_update" on public.reward_claims
  for update using (public.is_admin()) with check (public.is_admin());

-- 5) RPC : ajouter de l'XP (barème fixé côté serveur, anti-triche + anti-doublon)
create or replace function public.award_xp(p_action text, p_ref text default null)
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid uuid := auth.uid();
  v_xp  integer;
  v_ref text := nullif(p_ref, '');
  v_cur integer;
begin
  if v_uid is null then return null; end if;

  v_xp := case p_action
    when 'tournament_reg' then 100
    when 'team_create'    then 250
    when 'avatar_update'  then 20
    when 'checkin'        then 150
    when 'referral'       then 50
    else 0 end;
  if v_xp = 0 then
    return (select xp from public.profiles where id = v_uid);
  end if;

  -- Photo de profil : max 1 fois par mois
  if p_action = 'avatar_update' then
    v_ref := to_char(now(), 'YYYY-MM');
  end if;

  begin
    insert into public.xp_logs(user_id, action, xp, ref)
    values (v_uid, p_action, v_xp, v_ref);
  exception when unique_violation then
    return (select xp from public.profiles where id = v_uid);  -- déjà attribué
  end;

  update public.profiles set xp = coalesce(xp, 0) + v_xp
    where id = v_uid
    returning xp into v_cur;

  return v_cur;
end $$;

grant execute on function public.award_xp(text, text) to authenticated;

-- 6) RPC : parrainage (appelé par le NOUVEAU membre → crédite le parrain)
create or replace function public.claim_referral()
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid uuid := auth.uid();
  v_ref uuid;
begin
  if v_uid is null then return; end if;

  select referred_by into v_ref from public.profiles where id = v_uid;
  if v_ref is null or v_ref = v_uid then return; end if;
  if not exists (select 1 from public.profiles where id = v_ref) then return; end if;

  begin
    insert into public.xp_logs(user_id, action, xp, ref)
    values (v_ref, 'referral', 50, v_uid::text);
  exception when unique_violation then
    return;  -- filleul déjà compté
  end;

  update public.profiles set xp = coalesce(xp, 0) + 50 where id = v_ref;
end $$;

grant execute on function public.claim_referral() to authenticated;

-- 7) RPC : réclamer un cadeau (vérifie le palier XP)
create or replace function public.claim_reward(p_tier text)
returns public.reward_claims
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid  uuid := auth.uid();
  v_need integer;
  v_row  public.reward_claims;
begin
  if v_uid is null then raise exception 'auth required'; end if;

  v_need := case p_tier
    when 'xp_1000'  then 1000
    when 'xp_10000' then 10000
    when 'xp_20000' then 20000
    else null end;
  if v_need is null then raise exception 'bad tier'; end if;

  if (select coalesce(xp, 0) from public.profiles where id = v_uid) < v_need then
    raise exception 'not enough xp';
  end if;

  begin
    insert into public.reward_claims(user_id, tier, xp) values (v_uid, p_tier, v_need)
      returning * into v_row;
  exception when unique_violation then
    select * into v_row from public.reward_claims where user_id = v_uid and tier = p_tier;
  end;

  return v_row;
end $$;

grant execute on function public.claim_reward(text) to authenticated;

-- Anti-doublon
create unique index if not exists xp_logs_uniq_purchase     on public.xp_logs(user_id, ref) where action = 'purchase';
create unique index if not exists xp_logs_uniq_wallet_topup on public.xp_logs(user_id, ref) where action = 'wallet_topup';

-- Achat d'une commande -> +1000 XP (1 fois par commande)
create or replace function public.on_order_xp()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.user_id is null then return new; end if;
  begin
    insert into public.xp_logs(user_id, action, xp, ref)
    values (new.user_id, 'purchase', 1000, new.id::text);
  exception when unique_violation then
    return new;
  end;
  update public.profiles set xp = coalesce(xp, 0) + 1000 where id = new.user_id;
  return new;
end $$;

drop trigger if exists trg_order_xp on public.orders;
create trigger trg_order_xp after insert on public.orders
  for each row execute function public.on_order_xp();

-- Recharge wallet validée (>= 30 DT) -> +2500 XP (1 fois par recharge)
create or replace function public.on_recharge_xp()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.status = 'approved'
     and coalesce(old.status, '') <> 'approved'
     and coalesce(new.amount_tnd, 0) >= 30
     and new.user_id is not null then
    begin
      insert into public.xp_logs(user_id, action, xp, ref)
      values (new.user_id, 'wallet_topup', 2500, new.id::text);
    exception when unique_violation then
      return new;
    end;
    update public.profiles set xp = coalesce(xp, 0) + 2500 where id = new.user_id;
  end if;
  return new;
end $$;

drop trigger if exists trg_recharge_xp on public.recharges;
create trigger trg_recharge_xp after update on public.recharges
  for each row execute function public.on_recharge_xp();

-- ════════════════════════════════════════════════════════════════════
--  v3) RÉCOMPENSES : coût en XP + remboursement si l'admin refuse
--  Bronze  : 0 XP dépensé (badge)
--  Argent  : −9 000 XP (badge + 1 tournoi offert)
--  Or      : −19 000 XP (badge + cadeau Riot €/Steam…)
--  (Le seuil pour réclamer reste 1 000 / 10 000 / 20 000 XP)
--  À exécuter après le bloc v2 ci-dessus.
-- ════════════════════════════════════════════════════════════════════

alter table public.reward_claims add column if not exists cost integer not null default 0;
alter table public.reward_claims add column if not exists refunded boolean not null default false;

create or replace function public.claim_reward(p_tier text)
returns public.reward_claims
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid  uuid := auth.uid();
  v_need integer;
  v_cost integer;
  v_cur  integer;
  v_row  public.reward_claims;
begin
  if v_uid is null then raise exception 'auth required'; end if;

  v_need := case p_tier
    when 'xp_1000'  then 1000
    when 'xp_10000' then 10000
    when 'xp_20000' then 20000
    else null end;
  if v_need is null then raise exception 'bad tier'; end if;

  v_cost := case p_tier
    when 'xp_1000'  then 0
    when 'xp_10000' then 9000
    when 'xp_20000' then 19000
    else 0 end;

  select coalesce(xp, 0) into v_cur from public.profiles where id = v_uid for update;
  if v_cur is null then raise exception 'profile not found'; end if;
  if v_cur < v_need then raise exception 'not enough xp'; end if;

  select * into v_row from public.reward_claims
    where user_id = v_uid and tier = p_tier for update;

  if found then
    -- Déjà en attente ou validé : aucun second débit
    if v_row.status in ('pending', 'approved') then
      return v_row;
    end if;
    -- Refusé : nouvelle demande autorisée (on redébite, le remboursement précédent est annulé)
    update public.reward_claims
      set status = 'pending', cost = v_cost, refunded = false, decided_at = null, created_at = now()
      where id = v_row.id
      returning * into v_row;
    update public.profiles set xp = greatest(0, coalesce(xp, 0) - v_cost) where id = v_uid;
    return v_row;
  end if;

  insert into public.reward_claims(user_id, tier, xp, cost)
    values (v_uid, p_tier, v_need, v_cost)
    returning * into v_row;

  update public.profiles set xp = greatest(0, coalesce(xp, 0) - v_cost) where id = v_uid;

  return v_row;

exception when unique_violation then
  select * into v_row from public.reward_claims where user_id = v_uid and tier = p_tier;
  return v_row;
end $$;

grant execute on function public.claim_reward(text) to authenticated;

-- Décision admin : approuve/refuse et rembourse l'XP si la demande passe de pending -> rejected
create or replace function public.decide_reward(p_id uuid, p_status text)
returns public.reward_claims
language plpgsql
security definer
set search_path = public
as $$
declare
  v_row public.reward_claims;
begin
  if not public.is_admin() then raise exception 'admin only'; end if;
  if p_status not in ('approved', 'rejected') then raise exception 'bad status'; end if;

  select * into v_row from public.reward_claims where id = p_id for update;
  if not found then raise exception 'not found'; end if;

  if p_status = 'rejected' and v_row.status = 'pending' and not v_row.refunded then
    update public.profiles set xp = coalesce(xp, 0) + coalesce(v_row.cost, 0) where id = v_row.user_id;
    update public.reward_claims set refunded = true where id = p_id;
  end if;

  update public.reward_claims
    set status = p_status, decided_at = now()
    where id = p_id
    returning * into v_row;

  return v_row;
end $$;

grant execute on function public.decide_reward(uuid, text) to authenticated;

-- ════════════════════════════════════════════════════════════════════
--  v4) PROFIL PUBLIC — tout membre peut consulter le profil d'un autre
--  Expose UNIQUEMENT : nom, XP, VIP, Riot (pseudo/tag/région), badges validés.
--  Jamais : email, téléphone, solde, commandes, logs XP.
--  Le nom + avatar + bannière restent dans profile_assets (déjà public).
-- ════════════════════════════════════════════════════════════════════

create or replace function public.public_player(p_id uuid)
returns json
language plpgsql
security definer
set search_path = public
stable
as $$
declare
  v_badges json := '[]'::json;
  v_out json;
begin
  if to_regclass('public.reward_claims') is not null then
    select coalesce(json_agg(tier order by decided_at nulls last), '[]'::json)
      into v_badges
      from public.reward_claims
      where user_id = p_id and status = 'approved';
  end if;

  select json_build_object(
    'id', p.id,
    'name', coalesce(pa.name, p.name),
    'xp', coalesce(p.xp, 0),
    'is_vip', coalesce(p.is_vip, false),
    'vip_until', p.vip_until,
    'riot_game', p.riot_game,
    'riot_game_name', p.riot_game_name,
    'riot_tag_line', p.riot_tag_line,
    'riot_region', p.riot_region,
    'riot_linked_at', p.riot_linked_at,
    'badges', v_badges,
    'created_at', p.created_at
  ) into v_out
  from public.profiles p
  left join public.profile_assets pa on pa.id = p.id
  where p.id = p_id;

  return v_out;
end $$;

grant execute on function public.public_player(uuid) to anon, authenticated;
